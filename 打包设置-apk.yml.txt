name: 打包小周血压本 APK

on:
  push:
    paths:
      - 'android/**'
      - '.github/workflows/apk.yml'
  workflow_dispatch:

permissions:
  contents: write

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: '17'

      - uses: gradle/actions/setup-gradle@v4
        with:
          gradle-version: '8.9'

      - name: 打包 APK
        run: gradle -p android assembleRelease --no-daemon --stacktrace

      - name: 改名
        run: cp android/app/build/outputs/apk/release/app-release.apk xiaozhou.apk

      - name: 发布下载链接
        uses: softprops/action-gh-release@v2
        with:
          tag_name: apk
          name: 小周血压本 安卓 App
          body: 下载 xiaozhou.apk，在手机上点开安装。
          files: xiaozhou.apk
          make_latest: true

      - uses: actions/upload-artifact@v4
        with:
          name: xiaozhou-apk
          path: xiaozhou.apk
